cmake_minimum_required(VERSION 3.22)
project(gdal4k C CXX)
find_package(GDAL CONFIG QUIET)
if(NOT TARGET GDAL::GDAL)
    find_package(GDAL REQUIRED)
endif()
add_library(gdal4k SHARED gdal4k.cpp)
target_compile_features(gdal4k PRIVATE cxx_std_17)
target_compile_definitions(gdal4k PRIVATE GDAL4K_BUILD)
target_link_libraries(gdal4k PRIVATE GDAL::GDAL)
set_target_properties(gdal4k PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(WIN32)
    set_target_properties(gdal4k PROPERTIES PREFIX "")
endif()
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    find_package(JNI REQUIRED COMPONENTS JVM)
    target_sources(gdal4k PRIVATE jni.cpp)
    target_include_directories(gdal4k PRIVATE ${JNI_INCLUDE_DIRS})
elseif(ANDROID)
    target_sources(gdal4k PRIVATE jni.cpp)
    target_link_options(gdal4k PRIVATE "-Wl,-z,max-page-size=16384")
endif()

install(TARGETS gdal4k LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES gdal4k.h DESTINATION include)
