cmake_minimum_required(VERSION 3.22)
project(gdal4k_probe C CXX)
find_package(GDAL CONFIG QUIET)
if(NOT TARGET GDAL::GDAL)
    find_package(GDAL REQUIRED)
endif()
add_library(gdal4k_probe SHARED probe.c)
set_target_properties(gdal4k_probe PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON LINKER_LANGUAGE CXX)
if(WIN32)
    set_target_properties(gdal4k_probe PROPERTIES PREFIX "")
endif()
target_link_libraries(gdal4k_probe PRIVATE GDAL::GDAL)
if(NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    find_package(JNI REQUIRED COMPONENTS JVM)
    target_sources(gdal4k_probe PRIVATE jni.c)
    target_include_directories(gdal4k_probe PRIVATE ${JNI_INCLUDE_DIRS})
elseif(ANDROID)
    target_sources(gdal4k_probe PRIVATE jni.c)
endif()
